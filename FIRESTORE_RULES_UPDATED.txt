// ✅ Put this at the very top

rules_version = '2';

service cloud.firestore {

  // Helper: allow only vote/counter fields to change on a POST

  function postVoteOnlyUpdate() {

    return request.resource.data.diff(resource.data).changedKeys()

             .hasOnly(['upvotes','downvotes','commentCount','updatedAt'])

           && request.resource.data.authorId == resource.data.authorId;

  }

  // Helper: allow only vote fields to change on a COMMENT

  function commentVoteOnlyUpdate() {

    return request.resource.data.diff(resource.data).changedKeys()

             .hasOnly(['upvotes','downvotes','updatedAt'])

           && request.resource.data.authorId == resource.data.authorId;

  }

  match /databases/{database}/documents {

    // Users

    match /users/{userId} {

      allow read: if request.auth != null;

      allow write: if request.auth != null && request.auth.uid == userId;

      // ✅ Saved Prompts subcollection - users can only access their own saved prompts
      match /savedPrompts/{promptId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow create: if request.auth != null && request.auth.uid == userId;
        allow delete: if request.auth != null && request.auth.uid == userId;
        // No update needed - we only create/delete saved prompts, never update them
      }

    }

    // Posts

    match /posts/{postId} {

      // anyone signed in can read

      allow read: if request.auth != null;

      // create: authorId must be caller

      allow create: if request.auth != null

                    && request.resource.data.authorId == request.auth.uid;

      // update/delete:

      //   - author can do anything

      //   - non-authors may only change counters (upvotes/downvotes/commentCount/updatedAt)

      allow update, delete: if request.auth != null && (

        request.auth.uid == resource.data.authorId

        || postVoteOnlyUpdate()

      );

      // Per-user post votes (doc id == voter's uid)

      match /postVotes/{userId} {

        allow read: if request.auth != null;

        allow create, update, delete: if request.auth != null

                                      && request.auth.uid == userId;

      }

      // Comments under a post

      match /comments/{commentId} {

        allow read: if request.auth != null;

        // create: authorId must match caller

        allow create: if request.auth != null

                      && request.resource.data.authorId == request.auth.uid;

        // update/delete:

        //   - author can edit/delete own comment

        //   - non-authors may only change vote counters (up/down/updatedAt)

        allow update, delete: if request.auth != null && (

          request.auth.uid == resource.data.authorId

          || commentVoteOnlyUpdate()

        );

        // Per-user comment votes (doc id == voter's uid)

        match /commentVotes/{userId} {

          allow read: if request.auth != null;

          allow create, update, delete: if request.auth != null

                                        && request.auth.uid == userId;

        }

      }

    }

    // ✅ Prompts (for the Prompt Sharing page)

    match /prompts/{promptId} {

      // Signed-in users can read all prompts (set to 'true' if you want public read)

      allow read: if request.auth != null;

      // Create: only the owner; basic field checks

      allow create: if request.auth != null

                    && request.resource.data.authorId == request.auth.uid

                    && request.resource.data.text is string

                    && request.resource.data.text.size() > 0

                    && request.resource.data.text.size() <= 5000

                    // tags optional; if present must be a list

                    && (!('tags' in request.resource.data) || request.resource.data.tags is list);

      // Update: only the owner; keep basic constraints

      allow update: if request.auth != null

                    && request.auth.uid == resource.data.authorId

                    && request.resource.data.text is string

                    && request.resource.data.text.size() > 0

                    && request.resource.data.text.size() <= 5000

                    // If tags present, must remain a list

                    && (!('tags' in request.resource.data) || request.resource.data.tags is list);

      // Delete: only the owner

      allow delete: if request.auth != null

                    && request.auth.uid == resource.data.authorId;

    }

  }

}
